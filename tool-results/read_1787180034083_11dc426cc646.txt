     1→import { useRef, useEffect, useState, useCallback } from 'react';
     2→import {
     3→  ArrowLeft, Box, Plus, Move, RotateCw, Maximize, Paintbrush,
     4→  Copy, Trash2, MousePointer2, Hand, Info, Cuboid, Check, X
     5→} from 'lucide-react';
     6→
     7→/* ================================================================
     8→   3D BLOCK SIMULATOR — BABFTSS Style
     9→   Engine 3D from scratch using Canvas 2D (no external 3D libs)
    10→   ================================================================ */
    11→
    12→const GRID = 1;
    13→const GRID_SIZE = 14;
    14→const COLORS = [
    15→  '#3b82f6', '#ef4444', '#22c55e', '#f59e0b',
    16→  '#8b5cf6', '#ec4899', '#14b8a6', '#f97316',
    17→  '#64748b', '#1e293b', '#ffffff', '#84cc16'
    18→];
    19→
    20→class Vec3 {
    21→  constructor(x, y, z) { this.x = x; this.y = y; this.z = z; }
    22→  add(v) { return new Vec3(this.x + v.x, this.y + v.y, this.z + v.z); }
    23→  sub(v) { return new Vec3(this.x - v.x, this.y - v.y, this.z - v.z); }
    24→  mul(s) { return new Vec3(this.x * s, this.y * s, this.z * s); }
    25→  clone() { return new Vec3(this.x, this.y, this.z); }
    26→}
    27→
    28→const TOOLS = [
    29→  { id: 'place',  label: 'Place',  icon: Plus,         key: 'p' },
    30→  { id: 'move',   label: 'Move',   icon: Move,         key: 'm' },
    31→  { id: 'rotate', label: 'Rotate', icon: RotateCw,     key: 'r' },
    32→  { id: 'scale',  label: 'Scale',  icon: Maximize,     key: 's' },
    33→  { id: 'color',  label: 'Paint',  icon: Paintbrush,   key: 'c' },
    34→  { id: 'clone',  label: 'Clone',  icon: Copy,         key: 'k' },
    35→  { id: 'delete', label: 'Delete', icon: Trash2,       key: 'x' },
    36→];
    37→
    38→export default function BlockSimulator3D({ setPage }) {
    39→  const canvasRef = useRef(null);
    40→  const containerRef = useRef(null);
    41→
    42→  const [tool, setTool] = useState('place');
    43→  const [currentColor, setCurrentColor] = useState('#3b82f6');
    44→  const [blockCount, setBlockCount] = useState(0);
    45→  const [selectedInfo, setSelectedInfo] = useState(null);
    46→  const [showHelp, setShowHelp] = useState(false);
    47→  const [paintConfirm, setPaintConfirm] = useState(null);  // { block, newColor, originalColor, x, y } or null
    48→
    49→  const stateRef = useRef({
    50→    blocks: [],
    51→    selected: null,
    52→    // pitch MUST be negative for a "looking DOWN at grid" view.
    53→    // Positive pitch tilts the world's +Z axis downward on screen,
    54→    // which makes the grid floor project BELOW center and the scene
    55→    // look upside-down / reversed compared to standard 3D editors.
    56→    cam: { yaw: -0.75, pitch: -0.55, dist: 22, target: new Vec3(0, 0, 0) },
    57→    isOrbiting: false,
    58→    isTransforming: false,
    59→    dragStart: null,
    60→    camStart: null,
    61→    transformStart: null,
    62→    blockStart: null,
    63→    hoverGrid: null,   // posisi grid (Vec3) tempat ghost block akan digambar, null = tidak ada ghost
    64→    hoverColor: '#3b82f6',
    65→    dpr: 1,
    66→    W: 0, H: 0, cx: 0, cy: 0,
    67→  });
    68→
    69→  /* ---------- 3D Engine ---------- */
    70→  const rotY = (v, a) => {
    71→    const c = Math.cos(a), s = Math.sin(a);
    72→    return new Vec3(v.x * c - v.z * s, v.y, v.x * s + v.z * c);
    73→  };
    74→  const rotX = (v, a) => {
    75→    const c = Math.cos(a), s = Math.sin(a);
    76→    return new Vec3(v.x, v.y * c - v.z * s, v.y * s + v.z * c);
    77→  };
    78→
    79→  const project = useCallback((p) => {
    80→    const s = stateRef.current;
    81→    let v = p.sub(s.cam.target);
    82→    v = rotY(v, s.cam.yaw);
    83→    v = rotX(v, s.cam.pitch);
    84→    // Standard perspective projection: closer points = bigger.
    85→    // focalLength tunes overall scale; clamp denominator to avoid div-by-zero.
    86→    const focalLength = 700;
    87→    const scale = focalLength / Math.max(0.5, v.z + s.cam.dist);
    88→    return {
    89→      x: s.cx / s.dpr + v.x * scale,
    90→      y: s.cy / s.dpr - v.y * scale,
    91→      z: v.z,
    92→      scale
    93→    };
    94→  }, []);
    95→
    96→  const getBlockCorners = (b) => {
    97→    const sz = b.size || new Vec3(1, 1, 1);
    98→    const r = b.rot || new Vec3(0, 0, 0);
    99→    const corners = [
   100→      new Vec3(-0.5, -0.5, -0.5), new Vec3(0.5, -0.5, -0.5),
   101→      new Vec3(0.5, 0.5, -0.5),   new Vec3(-0.5, 0.5, -0.5),
   102→      new Vec3(-0.5, -0.5, 0.5),  new Vec3(0.5, -0.5, 0.5),
   103→      new Vec3(0.5, 0.5, 0.5),    new Vec3(-0.5, 0.5, 0.5)
   104→    ];
   105→    return corners.map(v => {
   106→      let p = new Vec3(v.x * sz.x, v.y * sz.y, v.z * sz.z);
   107→      p = rotY(p, r.y); p = rotX(p, r.x); p = rotY(p, r.z);
   108→      return b.pos.add(p);
   109→    });
   110→  };
   111→
   112→  const shadeColor = (hex, factor) => {
   113→    const h = hex.replace('#', '');
   114→    const r = parseInt(h.substr(0, 2), 16);
   115→    const g = parseInt(h.substr(2, 2), 16);
   116→    const b = parseInt(h.substr(4, 2), 16);
   117→    return `rgb(${Math.min(255, Math.max(0, Math.round(r * factor)))},${Math.min(255, Math.max(0, Math.round(g * factor)))},${Math.min(255, Math.max(0, Math.round(b * factor)))})`;
   118→  };
   119→
   120→  /* ---------- Render ---------- */
   121→  const render = useCallback(() => {
   122→    const canvas = canvasRef.current;
   123→    if (!canvas) return;
   124→    const ctx = canvas.getContext('2d');
   125→    const s = stateRef.current;
   126→
   127→    ctx.clearRect(0, 0, s.W / s.dpr, s.H / s.dpr);
   128→
   129→    // Background gradient (biar ada kedalaman, gak void hitam total) — pakai warna yang
   130→    // sudah dipakai di file ini juga (panelBg & bg halaman), bukan warna baru.
   131→    const w = s.W / s.dpr, h = s.H / s.dpr;
   132→    const bgGrad = ctx.createRadialGradient(w / 2, h / 2, 0, w / 2, h / 2, Math.max(w, h) * 0.7);
   133→    bgGrad.addColorStop(0, '#0e1420');
   134→    bgGrad.addColorStop(1, '#05080f');
   135→    ctx.fillStyle = bgGrad;
   136→    ctx.fillRect(0, 0, w, h);
   137→
   138→    // Grid
   139→    ctx.strokeStyle = 'rgba(148, 163, 184, 0.16)';
   140→    ctx.lineWidth = 0.5;
   141→    const N = GRID_SIZE;
   142→    for (let i = -N; i <= N; i++) {
   143→      const a = project(new Vec3(i * GRID, 0, -N * GRID));
   144→      const b = project(new Vec3(i * GRID, 0, N * GRID));
   145→      ctx.beginPath(); ctx.moveTo(a.x, a.y); ctx.lineTo(b.x, b.y); ctx.stroke();
   146→      const c = project(new Vec3(-N * GRID, 0, i * GRID));
   147→      const d = project(new Vec3(N * GRID, 0, i * GRID));
   148→      ctx.beginPath(); ctx.moveTo(c.x, c.y); ctx.lineTo(d.x, d.y); ctx.stroke();
   149→    }
   150→    // Axes
   151→    const o = project(new Vec3(0, 0, 0));
   152→    const xEnd = project(new Vec3(3.5, 0, 0));
   153→    const zEnd = project(new Vec3(0, 0, 3.5));
   154→    ctx.strokeStyle = '#ef4444'; ctx.lineWidth = 2;
   155→    ctx.beginPath(); ctx.moveTo(o.x, o.y); ctx.lineTo(xEnd.x, xEnd.y); ctx.stroke();
   156→    ctx.strokeStyle = '#3b82f6'; ctx.lineWidth = 2;
   157→    ctx.beginPath(); ctx.moveTo(o.x, o.y); ctx.lineTo(zEnd.x, zEnd.y); ctx.stroke();
   158→
   159→    // Blocks (painter's algorithm)
   160→    const sorted = s.blocks.map((b, i) => ({ b, i, depth: project(b.pos).z }))
   161→      .sort((a, b) => b.depth - a.depth);
   162→
   163→    sorted.forEach(item => {
   164→      const b = item.b;
   165→      const corners = getBlockCorners(b);
   166→      const pc = corners.map(project);
   167→      const faces = [
   168→        { idx: [3, 2, 1, 0], shade: 0.58 },
   169→        { idx: [4, 5, 6, 7], shade: 0.82 },
   170→        { idx: [0, 1, 5, 4], shade: 0.42 },
   171→        { idx: [7, 6, 2, 3], shade: 1.0 },
   172→        { idx: [4, 7, 3, 0], shade: 0.72 },
   173→        { idx: [1, 2, 6, 5], shade: 0.88 }
   174→      ];
   175→      faces.forEach(f => { f.avgZ = f.idx.reduce((sum, i2) => sum + pc[i2].z, 0) / 4; });
   176→      faces.sort((a, b2) => b2.avgZ - a.avgZ);
   177→
   178→      faces.forEach(f => {
   179→        const pts = f.idx.map(i2 => pc[i2]);
   180→        const ax = pts[1].x - pts[0].x, ay = pts[1].y - pts[0].y;
   181→        const bx = pts[2].x - pts[1].x, by = pts[2].y - pts[1].y;
   182→        if (ax * by - ay * bx < 0) return; // backface cull
   183→        ctx.beginPath();
   184→        ctx.moveTo(pts[0].x, pts[0].y);
   185→        for (let k = 1; k < pts.length; k++) ctx.lineTo(pts[k].x, pts[k].y);
   186→        ctx.closePath();
   187→        ctx.fillStyle = shadeColor(b.color, f.shade);
   188→        ctx.fill();
   189→        ctx.strokeStyle = 'rgba(0,0,0,0.18)';
   190→        ctx.lineWidth = 0.8;
   191→        ctx.stroke();
   192→      });
   193→
   194→      if (s.selected === b) {
   195→        ctx.strokeStyle = '#f472b6';
   196→        ctx.lineWidth = 2.5;
   197→        ctx.shadowColor = '#f472b6';
   198→        ctx.shadowBlur = 14;
   199→        ctx.beginPath();
   200→        ctx.moveTo(pc[0].x, pc[0].y);
   201→        for (let k = 1; k < 4; k++) ctx.lineTo(pc[k].x, pc[k].y);
   202→        ctx.closePath();
   203→        ctx.stroke();
   204→        ctx.shadowBlur = 0;
   205→      }
   206→    });
   207→
   208→    // Ghost/preview block — muncul saat mode Place & kursor di atas grid
   209→    if (s.hoverGrid) {
   210→      const ghost = { pos: s.hoverGrid, size: new Vec3(1, 1, 1), rot: new Vec3(0, 0, 0) };
   211→      const gCorners = getBlockCorners(ghost);
   212→      const gpc = gCorners.map(project);
   213→      const gFaces = [
   214→        { idx: [3, 2, 1, 0] },
   215→        { idx: [4, 5, 6, 7] },
   216→        { idx: [0, 1, 5, 4] },
   217→        { idx: [7, 6, 2, 3] },
   218→        { idx: [4, 7, 3, 0] },
   219→        { idx: [1, 2, 6, 5] }
   220→      ];
   221→      gFaces.forEach(f => { f.avgZ = f.idx.reduce((sum, i2) => sum + gpc[i2].z, 0) / 4; });
   222→      gFaces.sort((a, b2) => b2.avgZ - a.avgZ);
   223→
   224→      ctx.save();
   225→      ctx.globalAlpha = 0.35;
   226→      gFaces.forEach(f => {
   227→        const pts = f.idx.map(i2 => gpc[i2]);
   228→        const ax = pts[1].x - pts[0].x, ay = pts[1].y - pts[0].y;
   229→        const bx = pts[2].x - pts[1].x, by = pts[2].y - pts[1].y;
   230→        if (ax * by - ay * bx < 0) return; // backface cull, sama seperti block asli
   231→        ctx.beginPath();
   232→        ctx.moveTo(pts[0].x, pts[0].y);
   233→        for (let k = 1; k < pts.length; k++) ctx.lineTo(pts[k].x, pts[k].y);
   234→        ctx.closePath();
   235→        ctx.fillStyle = shadeColor(s.hoverColor, 1);
   236→        ctx.fill();
   237→      });
   238→      ctx.globalAlpha = 0.7;
   239→      ctx.strokeStyle = s.hoverColor;
   240→      ctx.lineWidth = 1.5;
   241→      ctx.setLineDash([4, 3]);
   242→      gFaces.forEach(f => {
   243→        const pts = f.idx.map(i2 => gpc[i2]);
   244→        const ax = pts[1].x - pts[0].x, ay = pts[1].y - pts[0].y;
   245→        const bx = pts[2].x - pts[1].x, by = pts[2].y - pts[1].y;
   246→        if (ax * by - ay * bx < 0) return;
   247→        ctx.beginPath();
   248→        ctx.moveTo(pts[0].x, pts[0].y);
   249→        for (let k = 1; k < pts.length; k++) ctx.lineTo(pts[k].x, pts[k].y);
   250→        ctx.closePath();
   251→        ctx.stroke();
   252→      });
   253→      ctx.setLineDash([]);
   254→      ctx.restore();
   255→    }
   256→  }, [project]);
   257→
   258→  /* ---------- Resize ---------- */
   259→  useEffect(() => {
   260→    const handleResize = () => {
   261→      const canvas = canvasRef.current;
   262→      const container = containerRef.current;
   263→      if (!canvas || !container) return;
   264→      const rect = container.getBoundingClientRect();
   265→      const dpr = window.devicePixelRatio || 1;
   266→      stateRef.current.dpr = dpr;
   267→      stateRef.current.W = canvas.width = rect.width * dpr;
   268→      stateRef.current.H = canvas.height = rect.height * dpr;
   269→      canvas.style.width = rect.width + 'px';
   270→      canvas.style.height = rect.height + 'px';
   271→      // Scale the 2D context so we can draw in CSS pixels directly.
   272→      // Without this, drawings land in buffer-pixel space and appear
   273→      // shrunk by `dpr` on retina displays.
   274→      const ctx = canvas.getContext('2d');
   275→      ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
   276→      stateRef.current.cx = stateRef.current.W / 2;
   277→      stateRef.current.cy = stateRef.current.H / 2;
   278→      render();
   279→    };
   280→    handleResize();
   281→    const container = containerRef.current;
   282→    const ro = container ? new ResizeObserver(handleResize) : null;
   283→    if (container && ro) ro.observe(container);
   284→    window.addEventListener('resize', handleResize);
   285→    return () => {
   286→      if (ro) ro.disconnect();
   287→      window.removeEventListener('resize', handleResize);
   288→    };
   289→  }, [render]);
   290→
   291→  /* ---------- Hit Test ---------- */
   292→  const pointInPoly = (x, y, poly) => {
   293→    let inside = false;
   294→    for (let i = 0, j = poly.length - 1; i < poly.length; j = i++) {
   295→      const xi = poly[i].x, yi = poly[i].y;
   296→      const xj = poly[j].x, yj = poly[j].y;
   297→      if (((yi > y) !== (yj > y)) && (x < (xj - xi) * (y - yi) / (yj - yi) + xi))
   298→        inside = !inside;
   299→    }
   300→    return inside;
   301→  };
   302→
   303→  const hitTest = (mx, my) => {
   304→    const s = stateRef.current;
   305→    for (let i = s.blocks.length - 1; i >= 0; i--) {
   306→      const b = s.blocks[i];
   307→      const pc = getBlockCorners(b).map(project);
   308→      const faces = [[0,1,2,3],[4,5,6,7],[0,1,5,4],[3,2,6,7],[0,3,7,4],[1,2,6,5]];
   309→      for (const f of faces) {
   310→        const pts = f.map(idx => pc[idx]);
   311→        if (pointInPoly(mx, my, pts)) return b;
   312→      }
   313→    }
   314→    return null;
   315→  };
   316→
   317→  const snap = (v) => new Vec3(Math.round(v.x), Math.round(v.y), Math.round(v.z));
   318→
   319→  const getGridPos = (mx, my) => {
   320→    let best = null, bestErr = 1e9;
   321→    for (let x = -GRID_SIZE; x <= GRID_SIZE; x += 0.5) {
   322→      for (let z = -GRID_SIZE; z <= GRID_SIZE; z += 0.5) {
   323→        const p = new Vec3(x, 0, z);
   324→        const pp = project(p);
   325→        const err = Math.hypot(pp.x - mx, pp.y - my);
   326→        if (err < bestErr) { bestErr = err; best = p; }
   327→      }
   328→    }
   329→    return best ? snap(best) : null;
   330→  };
   331→
   332→  /* ---------- Update UI State ---------- */
   333→  const updateUISelection = (b) => {
   334→    if (b) {
   335→      setSelectedInfo({
   336→        pos: `(${b.pos.x.toFixed(1)}, ${b.pos.y.toFixed(1)}, ${b.pos.z.toFixed(1)})`,
   337→        size: `(${b.size.x.toFixed(1)} x ${b.size.y.toFixed(1)} x ${b.size.z.toFixed(1)})`,
   338→        rot: `${(b.rot.y * 57.2958).toFixed(0)}deg`
   339→      });
   340→    } else {
   341→      setSelectedInfo(null);
   342→    }
   343→  };
   344→
   345→  /* ---------- Mouse Events ---------- */
   346→  useEffect(() => {
   347→    const canvas = canvasRef.current;
   348→    if (!canvas) return;
   349→
   350→    const getPlacementY = (gp) => {
   351→      const s = stateRef.current;
   352→      let y = 0.5;
   353→      const stack = s.blocks.filter(b =>
   354→        Math.abs(b.pos.x - gp.x) < 0.1 && Math.abs(b.pos.z - gp.z) < 0.1
   355→      );
   356→      if (stack.length > 0) {
   357→        stack.sort((a, b) => b.pos.y - a.pos.y);
   358→        y = stack[0].pos.y + (stack[0].size ? stack[0].size.y : 1);
   359→      }
   360→      return y;
   361→    };
   362→
   363→    const runPlace = (mx, my) => {
   364→      const s = stateRef.current;
   365→      const gp = getGridPos(mx, my);
   366→      if (!gp) return;
   367→      const y = getPlacementY(gp);
   368→      const nb = {
   369→        pos: new Vec3(gp.x, y, gp.z),
   370→        size: new Vec3(1, 1, 1),
   371→        rot: new Vec3(0, 0, 0),
   372→        color: currentColor,
   373→        id: Date.now() + Math.random()
   374→      };
   375→      s.blocks.push(nb);
   376→      s.selected = nb;
   377→      setBlockCount(s.blocks.length);
   378→      updateUISelection(nb);
   379→      render();
   380→    };
   381→
   382→    const onContextMenu = (e) => {
   383→      // Cegah menu klik-kanan bawaan browser muncul — klik-kanan dipakai untuk orbit kamera.
   384→      e.preventDefault();
   385→    };
   386→
   387→    const onMouseDown = (e) => {
   388→      const rect = canvas.getBoundingClientRect();
   389→      const mx = e.clientX - rect.left, my = e.clientY - rect.top;
   390→      const s = stateRef.current;
   391→
   392→      // Klik-kanan = orbit kamera, SELALU, di tool apa pun. (gaya Roblox Studio)
   393→      if (e.button === 2) {
   394→        s.isOrbiting = true;
   395→        s.dragStart = { x: mx, y: my };
   396→        s.camStart = { yaw: s.cam.yaw, pitch: s.cam.pitch };
   397→        s.hoverGrid = null; // sembunyikan ghost selama orbit
   398→        canvas.style.cursor = 'grabbing';
   399→        return;
   400→      }
   401→
   402→      if (e.button !== 0) return; // hanya klik-kiri yang lanjut ke logic tool di bawah
   403→
   404→      if (tool === 'place') {
   405→        runPlace(mx, my);
   406→        return;
   407→      }
   408→
   409→      const hit = hitTest(mx, my);
   410→      if (tool === 'move' || tool === 'rotate' || tool === 'scale') {
   411→        if (hit) {
   412→          s.selected = hit;
   413→          s.isTransforming = true;
   414→          s.transformStart = { x: mx, y: my };
   415→          s.blockStart = { pos: hit.pos.clone(), rot: hit.rot.clone(), size: hit.size.clone() };
   416→          updateUISelection(hit);
   417→        } else {
   418→          s.selected = null;
   419→          updateUISelection(null);
   420→        }
   421→        render();
   422→        return;
   423→      }
   424→
   425→      if (hit) {
   426→        s.selected = hit;
   427→        if (tool === 'delete') {
   428→          s.blocks = s.blocks.filter(b => b !== hit);
   429→          s.selected = null;
   430→          setBlockCount(s.blocks.length);
   431→          updateUISelection(null);
   432→        } else if (tool === 'clone') {
   433→          const nb = {
   434→            pos: hit.pos.add(new Vec3(1.2, 0, 0)),
   435→            size: new Vec3(hit.size.x, hit.size.y, hit.size.z),
   436→            rot: new Vec3(hit.rot.x, hit.rot.y, hit.rot.z),
   437→            color: hit.color,
   438→            id: Date.now() + Math.random()
   439→          };
   440→          s.blocks.push(nb);
   441→          s.selected = nb;
   442→          updateUISelection(nb);
   443→        } else if (tool === 'color') {
   444→          // Show confirmation popup instead of instantly applying color.
   445→          // User must click ✓ Confirm to apply, or Cancel to revert.
   446→          const originalColor = hit.color;
   447→          hit.color = currentColor;  // Apply temporarily for preview
   448→          render();
   449→          setPaintConfirm({
   450→            block: hit,
   451→            newColor: currentColor,
   452→            originalColor: originalColor,
   453→            x: mx,
   454→            y: my,
   455→          });
   456→        }
   457→      } else {
   458→        s.selected = null;
   459→        updateUISelection(null);
   460→      }
   461→      render();
   462→    };
   463→
   464→    const onMouseMove = (e) => {

   516→        render();
   517→      }
